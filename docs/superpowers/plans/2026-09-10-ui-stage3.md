# 설계형 종신보험 UI 단계 3 구현 계획 (추천 · 비교 · 인쇄 · 공유 · 설정)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 설계 캔버스 위에 제안 기능을 얹는다. `/start`에 니즈·HLV 추천 카드, `/compare`(같은 예산 3안 비교), `/print`(제안서 인쇄), `/s#…`(모바일 읽기 전용 공유 + 내 설계로 가져오기), `/settings`(가정 세트 선택·숫자 편집·envelope·골든 검산). 새 의존성 없음.

**Architecture:** 가정 세트와 envelope를 설계 상태(`DesignState.settings`)에 넣는다. 그러면 산출·검증·그래프 규칙·공유 링크·인쇄가 모두 같은 값을 쓰고, 설정 화면은 리듀서 액션 하나로 끝난다. 공유 링크는 상태 JSON을 브라우저 내장 `CompressionStream`으로 압축해 URL 해시에 싣는다. 읽기 전용 화면은 `DesignProvider`에 `initial`/`persist` prop을 주어 같은 컴포넌트를 재사용한다.

**Tech Stack:** 기존과 동일(Next.js 15, React 19, Tailwind v4, Recharts 3, vitest 5). Node 22와 최신 브라우저의 `CompressionStream`/`DecompressionStream`, `navigator.clipboard`, `window.print()`.

**결정 (2026-09-10 인터뷰, 계획서 v0.5 §0.7):** 추천은 `/start` 예산 섹션 위 카드 · 공유 링크는 읽기 전용 + "내 설계로 가져오기" · 제안서는 고객용 2쪽 + 설계사용 1쪽(체크박스로 끄기) · 설정은 가정 세트 선택 + 숫자 폼 편집 + 골든 G1 검산 표시.

**자체 결정:** 설정은 설계 상태의 일부(`settings`)로 저장·공유·인쇄된다(별도 저장 키 없음, 새 설계·초기화 시 이전 설정 유지) · 사용자 정의 가정 세트의 id는 `custom`, label에 원본 세트를 적는다 · 비교는 표준형(저해지 미적용) 보험료 기준으로 한다 · 공유 페이로드 `{ v: 1, app, state }`, 접두사 `d.`(deflate-raw) / `j.`(비압축) · 인쇄는 A4 세로, `@media print`로 상단 내비·버튼 숨김, 그래프는 고정 폭.

/** 산출 가정과 설계 제약. 설계 상태의 일부라서 공유 링크·인쇄에도 그대로 실린다 */
export interface Settings { assumption: AssumptionSet; envelope: EnvelopeParams }
export const DEFAULT_SETTINGS: Settings = { assumption: getAssumption(ASSUMPTION_ID), envelope: DEFAULT_ENVELOPE };
export const envelopeOf = (s: Pick<DesignState, "settings">): EnvelopeParams => s.settings?.envelope ?? DEFAULT_ENVELOPE;
export const assumptionOf = (s: Pick<DesignState, "settings">): AssumptionSet => s.settings?.assumption ?? DEFAULT_SETTINGS.assumption;

const num = (v: unknown, d: number) => (typeof v === "number" && Number.isFinite(v) ? v : d);
const numObj = <T extends Record<string, number>>(raw: unknown, d: T): T => {
  const o = (raw ?? {}) as Record<string, unknown>;
  return Object.fromEntries(Object.entries(d).map(([k, v]) => [k, num(o[k], v)])) as T;
};

/** 저장·공유에서 온 설정을 믿지 않는다. 숫자가 아니면 기본값, 알 수 없는 세트는 기본 세트 */
export function sanitizeSettings(raw: unknown): Settings {
  const r = (raw ?? {}) as Partial<Record<keyof Settings, unknown>>;
  const a = (r.assumption ?? {}) as Partial<AssumptionSet> & Record<string, unknown>;
  const base = ASSUMPTIONS.find((x) => x.id === a.id) ?? (a.id === "custom" ? DEFAULT_SETTINGS.assumption : undefined);
  let assumption = DEFAULT_SETTINGS.assumption;
  if (base) {
    const e = a.expenses as Record<string, unknown> | undefined;
    const expenses = e?.model === "simple"
      ? { model: "simple" as const, ...numObj(e, { alpha: 0.007, beta: 0.0015, gamma: 0.02 }) }
      : e?.model === "method"
        ? { model: "method" as const, ...numObj(e, { alphaS: 0.01, alphaP: 1, betaS: 0.0015, betaG: 0.045, betaPrime: 0.001, gamma: 0.025 }) }
        : base.expenses;
    assumption = {
      ...base,
      id: a.id === "custom" ? "custom" : base.id,
      label: typeof a.label === "string" ? a.label : base.label,
      interest: num(a.interest, base.interest), standardInterest: num(a.standardInterest, base.standardInterest),
      waiver: typeof a.waiver === "boolean" ? a.waiver : base.waiver,
      expenses,
      lowSurrender: numObj(a.lowSurrender, base.lowSurrender),
      needs: numObj(a.needs, base.needs),
    };
  }
  return { assumption, envelope: numObj(r.envelope, DEFAULT_ENVELOPE) };
}

/** 숫자를 하나라도 고치면 사용자 정의 세트가 된다 */
export function customize(a: AssumptionSet, baseId: string): AssumptionSet {
  const origin = a.id === "custom" ? baseId : a.id;
  return { ...a, id: "custom", version: "사용자 정의", label: `사용자 정의 (기본: ${origin})` };
}

/** 계획서 §3.5. 필요보장(니즈)과 인적자본(HLV)을 기준보험금 후보로 내고, 가족·부채·기존 보장으로 프리셋을 고른다 */
export function recommend(s: DesignState): Recommendation {
  const p = s.profile, a = assumptionOf(s), env = envelopeOf(s);
  const n = needs({ age: p.age, income: p.income, liquidAssets: p.liquidAssets, debt: p.debt, childrenAges: p.childrenAges, existingCover: p.groupCover + p.termCover }, a.needs);
  const cap = (x: number) => roundS0(Math.min(Math.max(x, S0_MIN), env.uwLimit));
  const presetId = recommendPreset({ age: p.age, childrenAges: p.childrenAges, debt: p.debt, groupCover: p.groupCover }, a.needs.independenceAge);
  return { needs: n, suggestedS0: cap(n.needs), hlvS0: cap(n.hlv), presetId, reason: `${PRESETS[presetId].label}: ${REASON[presetId](s)}` };
}

```ts
import { reducer, initialState, type DesignState } from "./state";

export const SHARE_VERSION = "0.1";      // 앱 버전. 불일치 시 경고만 하고 읽는다
const FORMAT = 1;                          // 페이로드 형식. 불일치 시 거부

export interface SharePayload { v: number; app: string; state: DesignState }

const toB64u = (bytes: Uint8Array) => {
  let bin = ""; for (let i = 0; i < bytes.length; i += 0x8000) bin += String.fromCharCode(...bytes.subarray(i, i + 0x8000));
  return btoa(bin).replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
};
const fromB64u = (s: string) => Uint8Array.from(atob(s.replace(/-/g, "+").replace(/_/g, "/")), (c) => c.charCodeAt(0));

async function pipe(bytes: Uint8Array, stream: CompressionStream | DecompressionStream): Promise<Uint8Array> {
  const res = new Response(new Blob([bytes]).stream().pipeThrough(stream as unknown as ReadableWritablePair<Uint8Array, Uint8Array>));
  return new Uint8Array(await res.arrayBuffer());
}

/** 설계 상태 → URL 해시 문자열. `d.` = deflate-raw 압축, `j.` = 비압축(구형 브라우저 폴백) */
export async function encodeShare(state: DesignState, compress = typeof CompressionStream !== "undefined"): Promise<string> {
  const json = new TextEncoder().encode(JSON.stringify({ v: FORMAT, app: SHARE_VERSION, state } satisfies SharePayload));
  if (!compress) return "j." + toB64u(json);
  return "d." + toB64u(await pipe(json, new CompressionStream("deflate-raw")));
}

/** 해시 문자열 → 페이로드. 형식이 다르거나 깨졌으면 null. 상태는 리듀서 load로 정화해서 돌려준다 */
export async function decodeShare(hash: string): Promise<SharePayload | null> {
  try {
    const h = hash.replace(/^#/, "");
    const body = h.slice(2);
    if (!body) return null;
    const bytes = h.startsWith("d.") ? await pipe(fromB64u(body), new DecompressionStream("deflate-raw")) : h.startsWith("j.") ? fromB64u(body) : null;
    if (!bytes) return null;
    const p = JSON.parse(new TextDecoder().decode(bytes)) as Partial<SharePayload>;
    if (p.v !== FORMAT || !p.state || typeof p.app !== "string") return null;
    return { v: FORMAT, app: p.app, state: reducer(initialState(), { type: "load", state: p.state }) };
  } catch {
    return null;
  }
}
```
